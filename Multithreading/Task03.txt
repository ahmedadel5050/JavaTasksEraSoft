/*Task 3 : 
 *  Print numbers 1 to 5 with a 1-second delay between each number using sleep().
 * */

public class Main {
	public static void main(String[] args) {
		Thread thread1 = new parallelRuns();
		thread1.start();


	}
}

//----------------------------------------------------------

public class parallelRuns extends Thread{
	@Override	
	public void run() {

		for(int i=1; i<6; i++ ) {
			System.out.println(i);
			try {
				Thread.sleep(1000);
			}catch(InterruptedException ex) {
				System.out.println("Thread InterruptedException");
			}
		}

	}
}
