/*Task 6 : 
 *  Create a BankAccount class shared between two types of threads
 * */

public class Main {
	public static void main(String[] args) {
        BankAccount account = new BankAccount();

        Depositor depositor = new Depositor(account);
        Withdrawer withdrawer = new Withdrawer(account);

        depositor.start();
        withdrawer.start();
	}
}

//-----------------------------------------------------------


class BankAccount {
    private int balance = 0;

    public synchronized void deposit(int amount) {
        balance += amount;
        System.out.println("Deposited: " + amount + " | Balance: " + balance);
        notify();
    }

    public synchronized void withdraw(int amount) {
        while (balance < amount) {
            try {
                System.out.println("Insufficient balance. Waiting...");
                wait(); 
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                System.out.println("Withdraw interrupted");
            }
        }
        balance -= amount;
        System.out.println("Withdrew: " + amount + " | Balance: " + balance);
    }
}

//-----------------------------------------------------------

class Depositor extends Thread {
    private BankAccount account;

    public Depositor(BankAccount account) {
        this.account = account;
    }

    @Override
    public void run() {
        try {
            for (int i = 0; i < 5; i++) {
                account.deposit(100);
                Thread.sleep(2000); 
            }
        } catch (InterruptedException e) {
            System.out.println("Depositor interrupted");
        }
    }
}


//-----------------------------------------------------------

class Withdrawer extends Thread {
    private BankAccount account;

    public Withdrawer(BankAccount account) {
        this.account = account;
    }

    @Override
    public void run() {
        try {
            for (int i = 0; i < 5; i++) { 
                account.withdraw(50);
                Thread.sleep(1000); 
            }
        } catch (InterruptedException e) {
            System.out.println("Withdrawer interrupted");
        }
    }
}
