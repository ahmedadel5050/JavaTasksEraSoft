import java.util.Arrays;
import java.util.List;
import java.util.Objects;
import java.util.stream.Collectors;

/*Task 29 : Remove null or empty strings from a list using stream. */

public class Main {
	public static void main(String[] args) {
		List<String> names = Arrays.asList("Ali", "Mona", "Ahmed", "Sara", "Amr", "Laila", "Kareem", "Nada", "Nour",
				"Samy", "", null);


		List<String> new_names = names.stream().filter(Objects::nonNull) 
				.filter(s -> !s.isEmpty())
				.collect(Collectors.toList());

		System.out.println(new_names);

	}
}
