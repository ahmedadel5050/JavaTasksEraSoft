import java.util.Arrays;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

/*Task 17 : Partition a list of numbers into even and odd using partitioningBy. */

public class Main {
	public static void main(String[] args) {
        List<Integer> numbers = Arrays.asList(10, 5, 3, 7, 2, 10, 5, 8, 9, 0, -3, 4);

        Map<Boolean, List<Integer>> parts = numbers.stream().collect(Collectors.partitioningBy(n -> n % 2 == 0));

        System.out.println("Even numbers: " + parts.get(true));
        System.out.println("Odd numbers: " + parts.get(false));

	}
}
