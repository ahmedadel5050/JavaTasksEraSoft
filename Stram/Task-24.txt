import java.util.Arrays;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.stream.Collectors;

/*Task 24 : Map a list of strings to their lengths */

public class Main {
	public static void main(String[] args) {
		List<String> names = Arrays.asList("Ali", "Mona", "Ahmed", "Sara", "Amr", "Laila", "Kareem", "Nada", "Nour",
				"Samy", "", null);

		Map<String, Integer> nameLengthMap = names.stream().filter(Objects::nonNull)
				.collect(Collectors.toMap(name -> name, String::length));

		System.out.println("Name-Length Map: " + nameLengthMap);

	}
}
