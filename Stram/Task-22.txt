import java.util.Arrays;
import java.util.List;
import java.util.Objects;
import java.util.stream.Collectors;

/*Task 22 : Extract all unique characters from a list of words. */

public class Main {
	public static void main(String[] args) {

		List<String> names = Arrays.asList("Ali", "Mona", "Ahmed", "Sara", "Amr", "Laila", "Kareem", "Nada", "Nour",
				"Samy", "", null);


		List<Character> uniqueChars = names.stream().filter(Objects::nonNull) 
				.filter(s -> !s.isEmpty())
				.flatMap(s -> s.chars().mapToObj(c -> (char) c)) 
				.map(Character::toLowerCase) 
				.distinct() 
				.sorted()
				.collect(Collectors.toList());

		System.out.println( uniqueChars);

	}
}
