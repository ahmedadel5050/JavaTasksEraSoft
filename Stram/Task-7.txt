import java.util.List;
import java.util.Arrays;
import java.util.Objects;
import java.util.Optional;;

/*Task 7 : Find the first element in a stream that matches a given condition. (starting with "a" and length > 3) */

public class Main {
	public static void main(String[] args) {
		List<String> names = Arrays.asList("Ali", "Mona", "Ahmed", "Sara", "Amr", "Laila", "Kareem", "Nada", "Nour",
				"Samy", "", null);


		Optional<String> result = names.stream().filter(Objects::nonNull) 
				.filter(name -> name.toLowerCase().startsWith("a")) 
				.filter(name -> name.length() > 3) 
				.findFirst(); 


		if (result.isPresent()) {
			System.out.println("First match: " + result.get());
		} else {
			System.out.println("No match found.");
		}

	}
}
