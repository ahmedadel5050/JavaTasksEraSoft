import java.util.List;
import java.util.Arrays;
import java.util.Objects;;

/*Task 6 : Count the number of strings longer than 5 characters*/

public class Main {
	public static void main(String[] args) {
		List<String> names = Arrays.asList("Ali", "Mona", "Ahmed", "Sara", "Amr", "Laila", "Kareem", "Nada", "Nour",
				"Samy", "", null);

		long count = names.stream().filter(Objects::nonNull) 
				.filter(name -> name.length() > 5) 
				.count(); 

		System.out.println("Number of strings longer than 5 characters: " + count);

	}
}
