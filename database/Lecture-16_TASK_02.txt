--create Employees table wiith column ( EmployeeID, FirstName, LastName, Department, Salary )
CREATE TABLE Employees(
EmployeeID  NUMBER(4) ,
FirstName   VARCHAR2(120) ,
LastName    VARCHAR2(120) ,
Department  VARCHAR2(120) ,
Salary      NUMBER(10,2)
);

--=====================================================
--Update the salary of an employee to 600000 with EmployeeID     101.
UPDATE Employees
SET salary = 600000
WHERE EmployeeID = 101 ;
COMMIT;
--=====================================================
--Delete a record of an employee who Department = 101.
DELETE FROM Employees
WHERE Department = '101' 
-- 0 rows deleted.
;
-- Delete a record of an employee who HAS same department like employee ID 101
DELETE FROM Employees
WHERE Department = (SELECT Department FROM Employees WHERE EmployeeID = 101) 
;
--=====================================================
--Retrieve all employees in the IT department.
select * from Employees
where Department ='IT';
